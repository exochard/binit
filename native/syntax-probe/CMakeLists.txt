cmake_minimum_required(VERSION 3.16)
project(binit_syntax_probe C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_path(BASH_INCLUDE_DIR config.h PATHS /usr/include/bash REQUIRED)
find_path(READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
add_library(binit_syntax_probe MODULE probe.c)
target_include_directories(binit_syntax_probe SYSTEM PRIVATE ${BASH_INCLUDE_DIR} ${BASH_INCLUDE_DIR}/include ${BASH_INCLUDE_DIR}/builtins ${READLINE_INCLUDE_DIR})
target_compile_options(binit_syntax_probe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
target_link_libraries(binit_syntax_probe PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(binit_syntax_probe PROPERTIES PREFIX "")
if(DEFINED ENV{BINIT_SYNTAX_RUNTIME} AND NOT "$ENV{BINIT_SYNTAX_RUNTIME}" STREQUAL "")
    set(SYNTAX_RUNTIME "$ENV{BINIT_SYNTAX_RUNTIME}")
    execute_process(COMMAND python3 -c "import sys; from pathlib import Path; sys.path.insert(0, sys.argv[1]); from fetch_syntax import verify_runtime; verify_runtime(Path(sys.argv[2]))" ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts ${SYNTAX_RUNTIME} RESULT_VARIABLE RUNTIME_CHECK)
    if(NOT RUNTIME_CHECK EQUAL 0)
        message(FATAL_ERROR "Pinned syntax runtime verification failed")
    endif()
    set(TREE_SITTER_FOUND TRUE)
    set(TREE_SITTER_LIBRARY ${SYNTAX_RUNTIME}/lib/libtree-sitter.a)
    set(TREE_SITTER_INCLUDE ${SYNTAX_RUNTIME}/source/lib/include)
else()
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(TREE_SITTER QUIET IMPORTED_TARGET tree-sitter)
        set(TREE_SITTER_LIBRARY PkgConfig::TREE_SITTER)
    endif()
endif()
if(TREE_SITTER_GRAMMAR AND TREE_SITTER_FOUND)
    add_executable(binit_tree_probe tree_probe.c)
    target_compile_options(binit_tree_probe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(TREE_SITTER_INCLUDE)
        target_include_directories(binit_tree_probe PRIVATE ${TREE_SITTER_INCLUDE})
    endif()
    target_link_libraries(binit_tree_probe PRIVATE ${TREE_SITTER_LIBRARY} bash_grammar)
endif()
if(TREE_SITTER_GRAMMAR)
    if(NOT TREE_SITTER_FOUND)
        message(FATAL_ERROR "The incremental parser experiment needs libtree-sitter")
    endif()
    add_library(bash_grammar SHARED ${TREE_SITTER_GRAMMAR}/src/parser.c ${TREE_SITTER_GRAMMAR}/src/scanner.c)
    target_include_directories(bash_grammar PRIVATE ${TREE_SITTER_GRAMMAR}/src)
endif()

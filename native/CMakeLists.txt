cmake_minimum_required(VERSION 3.16)
project(binit_completion C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_path(BASH_INCLUDE_DIR config.h PATHS /usr/include/bash /usr/local/include/bash REQUIRED)
find_path(READLINE_INCLUDE_DIR readline/readline.h REQUIRED)
add_library(binit_complete MODULE completion.c menu.c effects.c layout.c worker.c)
if(DEFINED ENV{BINIT_SYNTAX_SOURCE} AND NOT "$ENV{BINIT_SYNTAX_SOURCE}" STREQUAL "")
    set(SYNTAX_SOURCE "$ENV{BINIT_SYNTAX_SOURCE}")
    set(SYNTAX_RUNTIME "$ENV{BINIT_SYNTAX_RUNTIME}")
    if(SYNTAX_RUNTIME STREQUAL "")
        message(FATAL_ERROR "BINIT_SYNTAX_RUNTIME must name the prepared pinned runtime")
    endif()
    execute_process(COMMAND python3 ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/fetch_syntax.py --verify ${SYNTAX_SOURCE} --runtime-prefix ${SYNTAX_RUNTIME} RESULT_VARIABLE SYNTAX_CHECK)
    if(NOT SYNTAX_CHECK EQUAL 0)
        message(FATAL_ERROR "Pinned syntax source verification failed")
    endif()
    add_library(binit_grammar STATIC ${SYNTAX_SOURCE}/src/parser.c ${SYNTAX_SOURCE}/src/scanner.c)
    set_target_properties(binit_grammar PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(binit_grammar PRIVATE ${SYNTAX_SOURCE}/src)
    target_sources(binit_complete PRIVATE highlight.c)
    target_include_directories(binit_complete PRIVATE ${SYNTAX_RUNTIME}/source/lib/include)
    target_compile_definitions(binit_complete PRIVATE BINIT_HIGHLIGHT=1)
    target_link_libraries(binit_complete PRIVATE binit_grammar ${SYNTAX_RUNTIME}/lib/libtree-sitter.a)
    add_executable(test_highlight test_highlight.c highlight.c)
    target_include_directories(test_highlight PRIVATE ${SYNTAX_RUNTIME}/source/lib/include)
    target_compile_options(test_highlight PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    target_link_libraries(test_highlight PRIVATE binit_grammar ${SYNTAX_RUNTIME}/lib/libtree-sitter.a)
endif()
target_include_directories(binit_complete SYSTEM PRIVATE ${BASH_INCLUDE_DIR} ${BASH_INCLUDE_DIR}/include ${BASH_INCLUDE_DIR}/builtins ${READLINE_INCLUDE_DIR})
target_compile_options(binit_complete PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fvisibility=hidden -fstack-protector-strong)
target_link_options(binit_complete PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)
target_link_libraries(binit_complete PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(binit_complete PROPERTIES PREFIX "")

enable_testing()
add_library(binit_redisplay_test MODULE test_redisplay.c)
target_include_directories(binit_redisplay_test SYSTEM PRIVATE ${BASH_INCLUDE_DIR} ${BASH_INCLUDE_DIR}/include ${BASH_INCLUDE_DIR}/builtins ${READLINE_INCLUDE_DIR})
target_compile_options(binit_redisplay_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fvisibility=hidden)
set_target_properties(binit_redisplay_test PROPERTIES PREFIX "")
add_executable(test_menu test_menu.c menu.c)
target_compile_options(test_menu PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
add_test(NAME menu COMMAND test_menu)
add_executable(test_layout test_layout.c layout.c)
target_compile_options(test_layout PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
add_test(NAME layout COMMAND test_layout)
if(TARGET test_highlight)
    add_test(NAME highlight COMMAND test_highlight)
endif()
